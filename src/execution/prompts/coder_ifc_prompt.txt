import ifcopenshell
from ifcopenshell.api import run
import numpy as np
import math

# Helper to convert mm to meters
def m(mm_val): return mm_val / 1000.0

# 1. Project & Spatial hierarchy setup
model = ifcopenshell.file(schema="IFC4")
project = run("root.create_entity", model, ifc_class="IfcProject", name="STRATA Model")
run("unit.assign_unit", model)
context = run("context.add_context", model, context_type="Model")
body = run("context.add_context", model, context_type="Model", context_identifier="Body", target_view="MODEL_VIEW", parent=context)

site = run("root.create_entity", model, ifc_class="IfcSite", name="Site")
building = run("root.create_entity", model, ifc_class="IfcBuilding", name="Building")
run("aggregate.assign_object", model, relating_object=project, products=[site])
run("aggregate.assign_object", model, relating_object=site, products=[building])

storey = run("root.create_entity", model, ifc_class="IfcBuildingStorey", name="Ground Floor")
storey.Elevation = 0.0
run("aggregate.assign_object", model, relating_object=building, products=[storey])

# 2. Floor slabs (MUST define a closed loop of outer points)
slab = run("root.create_entity", model, ifc_class="IfcSlab", name="Ground Slab")
footprint = [(0.0, 0.0), (10.0, 0.0), (10.0, 5.0), (0.0, 5.0), (0.0, 0.0)]
slab_rep = run("geometry.add_slab_representation", model, context=body, polyline=footprint, depth=0.2)
run("geometry.assign_representation", model, product=slab, representation=slab_rep)
run("geometry.edit_object_placement", model, product=slab, matrix=np.eye(4))
run("spatial.assign_container", model, relating_structure=storey, products=[slab])

# 3. Walls
wall = run("root.create_entity", model, ifc_class="IfcWall", name="Wall A")
wall_rep = run("geometry.create_2pt_wall", model, element=wall, context=body,
    p1=(0.0, 0.0), p2=(10.0, 0.0), elevation=0.0, height=3.0, thickness=0.2)
run("geometry.assign_representation", model, product=wall, representation=wall_rep)
run("spatial.assign_container", model, relating_structure=storey, products=[wall])

# 4. Opening Helper for Doors & Windows
def create_opening_helper(model, context, storey, wall, wall_p1, wall_p2, offset_from_p1, width, height, sill_height, is_door):
    dx, dy = wall_p2[0] - wall_p1[0], wall_p2[1] - wall_p1[1]
    wall_len = math.hypot(dx, dy)
    ox, oy = wall_p1[0] + (dx / wall_len) * offset_from_p1, wall_p1[1] + (dy / wall_len) * offset_from_p1
    angle = math.atan2(dy, dx)
    
    opening = run("root.create_entity", model, ifc_class="IfcOpeningElement")
    rep = run("geometry.add_wall_representation", model, context=context, length=width, height=height, thickness=0.5)
    run("geometry.assign_representation", model, product=opening, representation=rep)
    run("feature.add_feature", model, feature=opening, element=wall)
    
    matrix = np.array([
        [math.cos(angle), -math.sin(angle), 0.0, ox],
        [math.sin(angle),  math.cos(angle), 0.0, oy],
        [0.0,              0.0,             1.0, sill_height],
        [0.0,              0.0,             0.0, 1.0]
    ])
    run("geometry.edit_object_placement", model, product=opening, matrix=matrix)
    
    filling = run("root.create_entity", model, ifc_class="IfcDoor" if is_door else "IfcWindow")
    run("feature.add_filling", model, opening=opening, element=filling)
    run("spatial.assign_container", model, relating_structure=storey, products=[filling])

# 5. Save
model.write("output/building.ifc")
